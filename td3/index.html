a. Informations pertinentes pour une carte

Pour chaque carte, il est pertinent de mémoriser :

  • Nom : Le nom de la carte.
  • Numéro : Le numéro de la carte dans le collecteur (1 à 135).
  • Puissance : Un entier entre 0 et 6.
  • Statistiques : Quatre valeurs pour l'amitié, l'aventure, la bravoure et la popularité.

b. Structure une_carte

typedef struct {
    char nom[50];
    int numero;
    int puissance;
    int amitie;
    int aventure;
    int bravoure;
    int popularite;
} une_carte;

c. Structure les_cartes

typedef struct {
    une_carte cartes[135];
} les_cartes;

d. Structure une_collection

typedef struct {
    int nb_exemplaires[135];
    une_carte cartes[135];
} une_collection;

e. Procédure cartes_manquantes

void cartes_manquantes(une_collection collection) {
    printf("Cartes manquantes :\n");
    for (int i = 0; i < 135; i++) {
        if (collection.nb_exemplaires[i] == 0) {
            une_carte carte = collection.cartes[i];
            printf("Nom: %s, Numero: %d, Puissance: %d, Amitié: %d, Aventure: %d, Bravoure: %d, Popularité: %d\n",
                   carte.nom, carte.numero, carte.puissance, carte.amitie, carte.aventure, carte.bravoure, carte.popularite);
        }
    }
}

f. Fonction exemplaires_multiples

int exemplaires_multiples(une_collection collection) {
    int count = 0;
    for (int i = 0; i < 135; i++) {
        if (collection.nb_exemplaires[i] > 1) {
            count++;
        }
    }
    return count;
}

g. Fonction collection_complete

int collection_complete(une_collection collection) {
    for (int i = 0; i < 135; i++) {
        if (collection.nb_exemplaires[i] == 0) {
            return 0; // Collection non complète
        }
    }
    return 1; // Collection complète
}

h. Fonction recherche_carte

int recherche_carte(une_collection collection, char *nom) {
    for (int i = 0; i < 135; i++) {
        if (strcmp(collection.cartes[i].nom, nom) == 0) {
            return i;
        }
    }
    return -1; // Carte non trouvée
}

i. Procédure reunir_collections

void reunir_collections(une_collection *c1, une_collection c2) {
    for (int i = 0; i < 135; i++) {
        c1->nb_exemplaires[i] += c2.nb_exemplaires[i];
    }
}

j. Structure une_pochette

typedef struct {
    int numeros[3]; // Numéros des cartes dans la pochette
} une_pochette;

k. Procédure ajout_pochette

void ajout_pochette(une_collection *collection, une_pochette pochette) {
    for (int i = 0; i < 3; i++) {
        int num = pochette.numeros[i] - 1;
        collection->nb_exemplaires[num]++;
    }
}

l. Procédure ajout_plusieurs_pochettes

void ajout_plusieurs_pochettes(une_collection *collection, une_pochette pochettes[], int n) {
    for (int i = 0; i < n; i++) {
        ajout_pochette(collection, pochettes[i]);
    }
}

m. Procédure retrait_carte

void retrait_carte(une_collection *collection, int numero) {
    if (collection->nb_exemplaires[numero - 1] > 0) {
        collection->nb_exemplaires[numero - 1]--;
    }
}

n. Fonction comparaison_collections

int comparaison_collections(une_collection c1, une_collection c2) {
    for (int i = 0; i < 135; i++) {
        if ((c1.nb_exemplaires[i] > 0 && c2.nb_exemplaires[i] == 0) ||
            (c1.nb_exemplaires[i] == 0 && c2.nb_exemplaires[i] > 0)) {
            return 0; // Différentes
        }
    }
    return 1; // Identiques
}

o. Fonction verification_faisabilite_echange

int verification_faisabilite_echange(une_collection c1, une_collection c2, int num1, int num2) {
    return c1.nb_exemplaires[num1 - 1] > 1 && c2.nb_exemplaires[num2 - 1] > 1;
}

p. Procédure echange_cartes

void echange_cartes(une_collection *c1, une_collection *c2, int num1, int num2) {
    if (verification_faisabilite_echange(*c1, *c2, num1, num2)) {
        c1->nb_exemplaires[num1 - 1]--;
        c2->nb_exemplaires[num2 - 1]--;
        c1->nb_exemplaires[num2 - 1]++;
        c2->nb_exemplaires[num1 - 1]++;
    }
}
